# 排序算法

## 直接插入排序（稳定）

```c
void InsertSort (SqList &L) {
    for(i = 2; i <= L.length; i++) {
        if (L.r[i].key < L.r[i-1].key) {
            L.r[0] = L.r[i];
            L.r[i] = L.r[i-1];
            for (j = i - 2; L.r[0].key < L.r[j].key; j++)
                L.r[j+1] = L.r[j];
            L.r[j+1] = L.r[0];
        }
    }
}
```

## 快速排序（不稳定）

```c
int Partition(SqList &L, int low, int high){
    L.r[0] = L.r[low];
    pivotkey = L.r[0].key;
    //将第一个元素设为枢纽，并存到下标0处
    while(low < high){
        while(low < high && L.r[high].key >= pivotkey)
            high--;
        //从high开始左找到第一个比枢纽小的元素
        L.r[low] = L.r[high];
        //将该元素移动到low
        while(low < high && L.r[low].key <= pivotkey)
            low++;
        //从low开始向右找到第一个比枢纽大的元素
        L.r[high] = L.r[low];
        //将该元素移动到high
    }
    L.r[low] = L.r[0];
    //此时枢纽位置low所指向的位置
    return low;//该枢纽最后的位置
}
void Qsort(SqList &L, int low, int high){
    if(low < high){
        pivotloc = Partition(L, low, high);
        pivotloc = Partition(L, low, pivotloc-1);
        pivotloc = Partition(L, pivotloc+1; high);
    }
}
void QuickSort (SqList &L) {
    QSort(L, 1, L.length);
}
```

## 冒泡排序（稳定）

```c
void BubbleSort (SqList &L) {
    change = true;
    for (i = 1; i <= L.length; i++) {
        change = false;
        for (j = 1; j <= L.length-i; j++) {
            if (L.r[j].key > L.r[j+1].key) {
                t = L.r[j].key;
                L.r[j].key = L.r[j+1].key;
                L.r[j+1].key = t;
            }
        }
    }
}
```
